cmake_minimum_required(VERSION 3.22)
project(regolith VERSION 0.1.0 LANGUAGES C CXX)

# Plain CMake: no ament, no colcon, no FetchContent. rclcpp is found like any other
# CMake package, and the install tree gets just enough ament index for `ros2 run`.
#
# Adapters (the ROS 2 bridge and the CAN bridge) are optional. Each has a real
# implementation and a *_disabled.cpp stub with the same functions, and the option
# chooses which one is linked, so app/main.cpp is identical in every configuration.
option(REGOLITH_WITH_ROS "Build the ROS 2 bridge (needs rclcpp)" ON)
option(REGOLITH_WITH_CAN "Build the SocketCAN bridge (Linux)" ON)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

set(BOX3D_SOURCE_DIR "$ENV{BOX3D_SOURCE_DIR}" CACHE PATH "Box3D v0.1.0 source tree")
if(NOT EXISTS "${BOX3D_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "BOX3D_SOURCE_DIR must point at a Box3D v0.1.0 source tree (nix-shell sets it)")
endif()
add_subdirectory(${BOX3D_SOURCE_DIR} ${CMAKE_BINARY_DIR}/box3d EXCLUDE_FROM_ALL)

find_package(Threads REQUIRED)
find_package(raylib REQUIRED)
find_package(EXPAT REQUIRED)

# -ffp-contract=off matches Box3D, so our float maths is reproducible across machines.
# ros/ros_bridge.cpp builds without -fno-exceptions and -fno-rtti, which rclcpp needs.
set(REGOLITH_FLAGS -Wall -Wextra -Wshadow -Wno-missing-field-initializers -ffp-contract=off)
set(REGOLITH_STRICT_FLAGS ${REGOLITH_FLAGS} -fno-exceptions -fno-rtti)

add_library(regolith_core STATIC
  core/actuator.cpp
  core/allocator.cpp
  core/shared_state.cpp
  core/common.cpp
  core/config.cpp
  core/config_file.cpp
  core/robot.cpp
  core/stl.cpp
  core/urdf.cpp
  core/terrain.cpp
  core/world.cpp
  core/worker_pool.cpp
  core/sensors/imu.cpp
  core/sensors/lidar.cpp
  core/sensors/mount.cpp
)
target_include_directories(regolith_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(regolith_core PUBLIC box3d::box3d Threads::Threads m PRIVATE EXPAT::EXPAT)
target_compile_options(regolith_core PRIVATE ${REGOLITH_STRICT_FLAGS})

# The viewer's font (JetBrains Mono), copied into the binary so nothing is loaded at run
# time. Without one the viewer falls back to Raylib's built-in font.
set(REGOLITH_FONT_REGULAR "$ENV{REGOLITH_FONT_REGULAR}" CACHE FILEPATH "TTF for the viewer's text")
set(REGOLITH_FONT_BOLD "$ENV{REGOLITH_FONT_BOLD}" CACHE FILEPATH "TTF for the viewer's headings")
set(ui_fonts_source ${CMAKE_BINARY_DIR}/generated/ui_fonts.cpp)
set(ui_fonts_text "// Generated by CMakeLists.txt from REGOLITH_FONT_REGULAR and REGOLITH_FONT_BOLD.\n#include <stdint.h>\n")
foreach(weight regular bold)
  string(TOUPPER ${weight} upper)
  set(font_path "${REGOLITH_FONT_${upper}}")
  if(font_path AND EXISTS "${font_path}")
    file(READ "${font_path}" font_hex HEX)
    string(LENGTH "${font_hex}" font_digits)
    math(EXPR font_size "${font_digits} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," font_bytes "${font_hex}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${font_path}")
  else()
    if(font_path)
      message(WARNING "REGOLITH_FONT_${upper}: ${font_path} does not exist")
    endif()
    set(font_size 0)
    set(font_bytes "0")
  endif()
  string(APPEND ui_fonts_text "extern const uint8_t ui_font_${weight}[] = {${font_bytes}};\nextern const uint64_t ui_font_${weight}_size = ${font_size};\n")
  message(STATUS "Regolith: viewer font (${weight}) ${font_size} bytes")
endforeach()
file(CONFIGURE OUTPUT ${ui_fonts_source} CONTENT "${ui_fonts_text}" @ONLY)

add_library(regolith_render STATIC render/viewer.cpp render/sensor_camera.cpp render/ui.cpp ${ui_fonts_source})
target_link_libraries(regolith_render PUBLIC regolith_core raylib)
target_compile_options(regolith_render PRIVATE ${REGOLITH_STRICT_FLAGS})

if(REGOLITH_WITH_ROS)
  find_package(rclcpp REQUIRED)
  find_package(rosgraph_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  # The one place rclcpp is included; it needs exceptions and RTTI.
  add_library(regolith_ros STATIC ros/ros_bridge.cpp)
  target_link_libraries(regolith_ros PUBLIC regolith_core PRIVATE rclcpp::rclcpp ${rosgraph_msgs_TARGETS} ${std_msgs_TARGETS} ${sensor_msgs_TARGETS})
  target_compile_options(regolith_ros PRIVATE ${REGOLITH_FLAGS})
else()
  add_library(regolith_ros STATIC ros/ros_disabled.cpp)
  target_link_libraries(regolith_ros PUBLIC regolith_core)
  target_compile_options(regolith_ros PRIVATE ${REGOLITH_STRICT_FLAGS})
endif()
message(STATUS "Regolith: ROS 2 bridge ${REGOLITH_WITH_ROS}")

# The devices' wire formats (BLCMDs, LED strip) are plain code, always built and tested;
# only the socket is optional.
add_library(regolith_can_protocol STATIC can/blcmd.cpp can/led_strip.cpp)
target_link_libraries(regolith_can_protocol PUBLIC regolith_core)
target_compile_options(regolith_can_protocol PRIVATE ${REGOLITH_STRICT_FLAGS})
if(REGOLITH_WITH_CAN)
  add_library(regolith_can STATIC can/can_bridge.cpp)
  target_link_libraries(regolith_can PUBLIC regolith_core PRIVATE regolith_can_protocol)
else()
  add_library(regolith_can STATIC can/can_disabled.cpp)
  target_link_libraries(regolith_can PUBLIC regolith_core)
endif()
target_compile_options(regolith_can PRIVATE ${REGOLITH_STRICT_FLAGS})
message(STATUS "Regolith: CAN bridge ${REGOLITH_WITH_CAN}")

add_executable(regolith app/main.cpp)
target_link_libraries(regolith PRIVATE regolith_core regolith_render regolith_ros regolith_can)
target_compile_options(regolith PRIVATE ${REGOLITH_STRICT_FLAGS})

enable_testing()
foreach(suite shared_state config robot urdf world blcmd led_strip sensors)
  add_executable(test_${suite} tests/test_${suite}.cpp)
  target_link_libraries(test_${suite} PRIVATE regolith_core regolith_can_protocol)
  target_compile_options(test_${suite} PRIVATE ${REGOLITH_STRICT_FLAGS})
endforeach()
add_test(NAME shared_state COMMAND test_shared_state)
add_test(NAME config COMMAND test_config ${CMAKE_CURRENT_SOURCE_DIR}/config/sim.yaml)
add_test(NAME robot COMMAND test_robot ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/banksia.urdf)
add_test(NAME urdf COMMAND test_urdf ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/banksia.urdf)
add_test(NAME world COMMAND test_world)
add_test(NAME blcmd COMMAND test_blcmd)
add_test(NAME led_strip COMMAND test_led_strip)
add_test(NAME sensors COMMAND test_sensors ${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/banksia.urdf)

install(TARGETS regolith RUNTIME DESTINATION lib/regolith)
install(DIRECTORY config DESTINATION share/regolith)
if(REGOLITH_WITH_ROS)
  # Laid out like an ament package so `ros2 run regolith regolith` and
  # `ros2 launch regolith ...` find it.
  install(FILES package.xml DESTINATION share/regolith)
  install(DIRECTORY launch DESTINATION share/regolith)
  file(WRITE ${CMAKE_BINARY_DIR}/ament_index_marker "")
  install(FILES ${CMAKE_BINARY_DIR}/ament_index_marker
    DESTINATION share/ament_index/resource_index/packages RENAME regolith)
endif()
